    ```tsx
    'use client';

    import { useState, useEffect, useCallback } from 'react';
    import { useRouter } from 'next/navigation';
    import { Button } from '@/components/ui/Button';
    import { createOrder } from '@/lib/orders/actions';

    interface PaymentModalProps {
      isOpen: boolean;
      planName: string;
      amount: number;
      onCancel: () => void;
    }

    export function PaymentModal({
      isOpen,
      planName,
      amount,
      onCancel,
    }: PaymentModalProps) {
      const router = useRouter();
      const [loading, setLoading] = useState(false);
      const [error, setError] = useState<string | null>(null);

      const handleKeyDown = useCallback(
        (e: KeyboardEvent) => {
          if (e.key === 'Escape') {
            onCancel();
          }
        },
        [onCancel]
      );

      useEffect(() => {
        if (isOpen) {
          document.addEventListener('keydown', handleKeyDown);
          return () => document.removeEventListener('keydown', handleKeyDown);
        }
      }, [isOpen, handleKeyDown]);

      const handleConfirm = async () => {
        setLoading(true);
        setError(null);

        const result = await createOrder(planName, amount);

        if ('error' in result) {
          setError('订单创建失败，请重试');
          setLoading(false);
          return;
        }

        router.push(`/payment-success?id=${result.orderId}`);
      };

      if (!isOpen) return null;

      return (
        <div
          className="fixed inset-0 z-50 flex items-center justify-center p-4"
          role="dialog"
          aria-modal="true"
          aria-label="确认支付"
        >
          {/* Backdrop -- does NOT dismiss on click */}
          <div className="absolute inset-0 bg-black/50" />

          {/* Modal card */}
          <div className="relative bg-white rounded-xl shadow-xl max-w-sm w-full p-6">
            <h3 className="text-lg font-bold text-gray-900 text-center">确认支付</h3>

            <div className="mt-6 space-y-3">
              <div className="flex justify-between text-sm">
                <span className="text-gray-500">套餐</span>
                <span className="text-gray-900 font-bold">{planName}</span>
              </div>
              <div className="flex justify-between text-sm">
                <span className="text-gray-500">金额</span>
                <span className="text-gray-900 font-bold text-lg">
                  ¥{amount.toLocaleString()}
                </span>
              </div>
            </div>

            {/* Simulated payment notice */}
            <div className="mt-2 p-3 bg-amber-50 rounded-lg text-xs text-amber-700">
              当前为模拟支付，不会产生实际扣款。
              正式上线后将接入微信支付/支付宝。
            </div>

            {/* Error state */}
            {error && (
              <p className="mt-2 text-sm text-red-600 text-center">{error}</p>
            )}

            {/* Actions */}
            <div className="mt-6 space-y-3">
              <Button variant="primary" onClick={handleConfirm} loading={loading} autoFocus>
                确认支付 ¥{amount.toLocaleString()}
              </Button>
              <Button variant="ghost" onClick={onCancel} disabled={loading}>
                取消
              </Button>
            </div>
          </div>
        </div>
      );
    }
    ```

    Per UI-SPEC:
    - 'use client' component rendered as conditional overlay.
    - Props: isOpen, planName, amount, onCancel (confirm logic is internal via createOrder import).
    - Backdrop click on the outer div does NOT dismiss (no onClick handler on backdrop div -- prevents accidental close during payment flow per UI-SPEC).
    - Escape key dismisses via keydown event listener (attached in useEffect, cleaned up on unmount).
    - Confirm button has autoFocus (first focus on open per accessibility contract).
    - On confirm: calls createOrder Server Action imported from @/lib/orders/actions.
    - On success: router.push('/payment-success?id=...') per D-04 flow.
    - On failure: setError with "订单创建失败，请重试", modal stays open for retry.
    - Simulated payment notice: bg-amber-50 text-xs text-amber-700 with rounded corners.
    - "取消" button disabled while loading to prevent double-trigger.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| browser -> PaymentModal | User clicks confirm in the modal, triggering client-side createOrder Server Action call |
| PaymentModal -> createOrder Server Action | planName and amount cross from client to authenticated server endpoint |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-03-01 | Tampering | PaymentModal -> createOrder input | accept (MVP) | planName and amount originate from the server-rendered PLANS constant. However, a client could modify these before the Server Action call. createOrder validates planName as non-empty string and amount as positive integer, but does not validate against known plan prices. This is acceptable for MVP simulated payments (no real money). For production, server-side price validation against a plans table is required. |
| T-04-03-02 | Spoofing | PaymentModal auth bypass | mitigate | createOrder calls `supabase.auth.getUser()` internally. Unauthenticated users cannot create orders -- the Server Action returns error before any DB write. |
| T-04-03-03 | Elevation of Privilege | client-side price modification | accept (MVP) | A user who modifies the amount in browser devtools before calling createOrder could record a non-standard amount. Since this is simulated payment with no real money (per D-03), this is acceptable for MVP. Tracked as future security concern when real payment integration happens. |
</threat_model>
