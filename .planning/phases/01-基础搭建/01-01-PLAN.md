    **src/lib/supabase/client.ts** -- Browser client (Client Components only):
    ```typescript
    import { createBrowserClient } from '@supabase/ssr'

    export function createClient() {
      return createBrowserClient(
        process.env.NEXT_PUBLIC_SUPABASE_URL!,
        process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!
      )
    }
    ```

    **src/lib/supabase/server.ts** -- Server client (Server Components, Server Actions, Route Handlers):
    ```typescript
    import { createServerClient } from '@supabase/ssr'
    import { cookies } from 'next/headers'

    export async function createClient() {
      const cookieStore = await cookies()

      return createServerClient(
        process.env.NEXT_PUBLIC_SUPABASE_URL!,
        process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
        {
          cookies: {
            getAll() {
              return cookieStore.getAll()
            },
            setAll(cookiesToSet) {
              try {
                cookiesToSet.forEach(({ name, value, options }) => {
                  cookieStore.set(name, value, options)
                })
              } catch {
                // The `try/catch` is expected -- Server Components cannot set cookies.
                // Middleware handles actual cookie persistence.
              }
            },
          },
        }
      )
    }
    ```

    **src/lib/supabase/middleware.ts** -- Session update helper (called by middleware.ts):
    ```typescript
    import { createServerClient } from '@supabase/ssr'
    import { NextResponse, type NextRequest } from 'next/server'

    export async function updateSession(request: NextRequest) {
      let response = NextResponse.next({ request: { headers: request.headers } })

      const supabase = createServerClient(
        process.env.NEXT_PUBLIC_SUPABASE_URL!,
        process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
        {
          cookies: {
            getAll() {
              return request.cookies.getAll()
            },
            setAll(cookiesToSet) {
              cookiesToSet.forEach(({ name, value, options }) => {
                response.cookies.set(name, value, options)
              })
            },
          },
        }
      )

      // CRITICAL: getUser() refreshes expired access tokens via refresh token.
      // Without this, session is lost on page refresh.
      const {
        data: { user },
      } = await supabase.auth.getUser()

      return { response, user }
    }
    ```

    **src/middleware.ts** -- Next.js middleware entry point:
    ```typescript
    import { type NextRequest } from 'next/server'
    import { updateSession } from '@/lib/supabase/middleware'

    export async function middleware(request: NextRequest) {
      const { response, user } = await updateSession(request)

      // Protected routes (Phase 2+ will add /dashboard, /chat, etc.)
      if (request.nextUrl.pathname.startsWith('/dashboard') && !user) {
        return NextResponse.redirect(new URL('/login', request.url))
      }

      // Redirect authenticated users away from auth pages
      if (
        user &&
        (request.nextUrl.pathname.startsWith('/login') ||
         request.nextUrl.pathname.startsWith('/reset-password'))
      ) {
        return NextResponse.redirect(new URL('/dashboard', request.url))
      }

      return response
    }

    export const config = {
      matcher: [
        '/((?!_next/static|_next/image|favicon.ico|.*\\.(?:svg|png|jpg|jpeg|gif|webp)$).*)',
      ],
    }
    ```

    Key constraints from research anti-patterns:
    - middleware.ts MUST call `getUser()` (not `getSession()`) to verify JWT with Supabase server
    - server.ts setAll is wrapped in try/catch because Server Components cannot set cookies
    - middleware.ts setAll sets cookies on the response (this is where persistence actually happens)

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser -> Supabase Auth | User credentials (email/password) transmitted over HTTPS |
| Middleware -> Supabase Auth | JWT verification via getUser() on every request |
| Server Actions -> Supabase Auth | Auth operations re-verify session independently |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-01 | Information Disclosure | Environment variables | mitigate | .env.local in .gitignore; only NEXT_PUBLIC_* vars exposed to client; anon key has RLS enforced |
| T-01-02 | Spoofing | Session cookies | mitigate | httpOnly cookies via @supabase/ssr; Secure + SameSite attributes; getUser() verifies JWT with Supabase server (not trusting cookie data) |
| T-01-03 | Tampering | Middleware route protection | mitigate | Middleware runs before rendering; unauthenticated requests to /dashboard redirected to /login before any page logic executes |
| T-01-04 | Tampering | Server Actions auth | accept | Will be addressed in Plan 02 when Server Actions are created; each Server Action will independently re-verify auth |
</threat_model>
